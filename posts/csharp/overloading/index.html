class Program
{
	static int Add(int a, int b)
	{
		return a+b;
	}

	static int Add(int a, int b, int c)
	{
		return a+b+c;
	}
	
	static void Main(string[] args)
	{
		int a = 5;
		int b = 3;
		int c = 4;
		Console.WriteLine(a,b);
		Console.WriteLine(a,b,c);
	}
}
